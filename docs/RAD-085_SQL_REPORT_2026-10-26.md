من جذر المستودع بعد npm ci، في بيئة Linux x64 وبالاعتمادات الخارجية المؤقتة فقط:

python3 - <<'SETUP'

for alias, target in [('libicuuc.so.60', 'libicuuc.so.60.2'), ('libicudata.so.60', 'libicudata.so.60.2'), ('libicui18n.so.60', 'libicui18n.so.60.2'), ('libpq.so.5', 'libpq.so.5.18')]:
